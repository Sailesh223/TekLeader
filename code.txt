// ─── Circular Progress Ring Component ───────────────────────
interface ProgressRingProps {
  value: number;
  max: number;
  size?: number;
  strokeWidth?: number;
  color: string;
  glowColor: string;
  label: string;
  icon: React.ReactNode;
}

function ProgressRing({
  value,
  max,
  size = 120,
  strokeWidth = 8,
  color,
  glowColor,
  label,
  icon,
}: ProgressRingProps) {
  const radius = (size - strokeWidth) / 2;
  const circumference = radius * 2 * Math.PI;
  const percent = Math.min((value / max) * 100, 100);
  const strokeDashoffset = circumference - (percent / 100) * circumference;

  return (
    <Box sx={{ position: 'relative', display: 'inline-flex', flexDirection: 'column', alignItems: 'center' }}>
      <Box sx={{ position: 'relative', width: size, height: size }}>
        {/* Background glow */}
        <Box
          sx={{
            position: 'absolute',
            inset: -4,
            borderRadius: '50%',
            background: `radial-gradient(circle, ${glowColor}20 0%, transparent 70%)`,
          }}
        />
        <svg width={size} height={size} style={{ transform: 'rotate(-90deg)' }}>
          {/* Track */}
          <circle
            cx={size / 2}
            cy={size / 2}
            r={radius}
            fill="none"
            stroke="rgba(255,255,255,0.08)"
            strokeWidth={strokeWidth}
          />
          {/* Progress */}
          <circle
            cx={size / 2}
            cy={size / 2}
            r={radius}
            fill="none"
            stroke={`url(#ring-gradient-${label.replace(/\s/g, '')})`}
            strokeWidth={strokeWidth}
            strokeLinecap="round"
            strokeDasharray={circumference}
            strokeDashoffset={strokeDashoffset}
            style={{
              transition: 'stroke-dashoffset 1.5s cubic-bezier(0.4, 0, 0.2, 1)',
              filter: `drop-shadow(0 0 6px ${glowColor}80)`,
            }}
          />
          <defs>
            <linearGradient id={`ring-gradient-${label.replace(/\s/g, '')}`} x1="0%" y1="0%" x2="100%" y2="100%">
              <stop offset="0%" stopColor={color} />
              <stop offset="100%" stopColor={glowColor} />
            </linearGradient>
          </defs>
        </svg>
        {/* Center content */}
        <Box
          sx={{
            position: 'absolute',
            inset: 0,
            display: 'flex',
            flexDirection: 'column',
            alignItems: 'center',
            justifyContent: 'center',
          }}
        >
          <Box sx={{ color: color, mb: 0.3 }}>{icon}</Box>
          <Typography
            sx={{
              fontWeight: 900,
              fontSize: '1.3rem',
              color: '#fff',
              lineHeight: 1,
              animation: `${countUp} 0.8s ease-out`,
            }}
          >
            {typeof value === 'number' && value % 1 !== 0 ? value.toFixed(1) : value}
          </Typography>
        </Box>
      </Box>
      <Typography
        sx={{
          mt: 1,
          fontSize: '0.7rem',
          fontWeight: 700,
          textTransform: 'uppercase',
          letterSpacing: '1.5px',
          color: color,
        }}
      >
        {label}
      </Typography>
    </Box>
  );
}

// ─── Custom Dark Tooltip ────────────────────────────────────
function CustomTooltip({ active, payload, label, accentColor = '#00BFA5' }: any) {
  if (!active || !payload?.length) return null;
  return (
    <Box
      sx={{
        background: 'linear-gradient(135deg, #1a1a2e 0%, #16213e 100%)',
        border: `1px solid ${accentColor}60`,
        borderRadius: 2,
        p: 1.5,
        boxShadow: `0 8px 32px rgba(0,0,0,0.4), 0 0 20px ${accentColor}20`,
        backdropFilter: 'blur(20px)',
        minWidth: 150,
      }}
    >
      <Typography sx={{ fontWeight: 700, color: accentColor, fontSize: '0.8rem', mb: 0.5 }}>
        {label}
      </Typography>
      {payload.map((entry: any, i: number) => (
        <Box key={i} sx={{ display: 'flex', alignItems: 'center', gap: 1, mt: 0.3 }}>
          <Box sx={{ width: 8, height: 8, borderRadius: '50%', background: entry.color, boxShadow: `0 0 6px ${entry.color}` }} />
          <Typography sx={{ color: 'rgba(255,255,255,0.7)', fontSize: '0.75rem' }}>
            {entry.name}:
          </Typography>
          <Typography sx={{ color: '#fff', fontWeight: 700, fontSize: '0.75rem' }}>
            {entry.value}
          </Typography>
        </Box>
      ))}
    </Box>
  );
}

export default function AnalyticsPage() {
  const [loading, setLoading] = useState(true);
  const [selectedMonth, setSelectedMonth] = useState('');
  const [availableMonths, setAvailableMonths] = useState<string[]>([]);
  const [stats, setStats] = useState({
    totalManagers: 0,
    avgScore: 0,
    goldCount: 0,
    silverCount: 0,
    bronzeCount: 0,
    ignitionCount: 0,
  });
  const [utilizationData, setUtilizationData] = useState<any[]>([]);
  const [bandTrendData, setBandTrendData] = useState<any[]>([]);
  const [scoreTrendData, setScoreTrendData] = useState<any[]>([]);
  const [badgeTrendData, setBadgeTrendData] = useState<any[]>([]);

  useEffect(() => {
    loadMonths();
  }, []);

  useEffect(() => {
    if (selectedMonth) loadAnalytics();
  }, [selectedMonth]);

  useEffect(() => {
    if (availableMonths.length > 0) loadHistoricalData();
  }, [availableMonths]);

  const loadMonths = async () => {
    try {
      const response = await leaderboardApi.getAvailableMonths();
      setAvailableMonths(response.months);
      if (response.latestMonth) setSelectedMonth(response.latestMonth);
    } catch (error) {
      console.error('Failed to load months:', error);
    }
  };

  const loadHistoricalData = async () => {
    try {
      const historicalBandData: any[] = [];
      const historicalScoreData: any[] = [];
      const historicalBadgeData: any[] = [];

      for (const month of availableMonths) {
        const response = await leaderboardApi.getLeaderboard(month, 'all', 'all', '', 0, 1000);
        const managers = response.managers;

        const goldCount = managers.filter((m) => m.classificationBand === 'Gold').length;
        const silverCount = managers.filter((m) => m.classificationBand === 'Silver').length;
        const bronzeCount = managers.filter((m) => m.classificationBand === 'Bronze').length;
        const ignitionCount = managers.filter((m) => m.classificationBand === 'Ignition Zone').length;
        const avgScore = managers.reduce((sum, m) => sum + m.finalScore, 0) / managers.length;
        const totalBadges = managers.reduce((sum, m) => sum + (m.badges?.length || 0), 0);

        historicalBandData.push({ month, Gold: goldCount, Silver: silverCount, Bronze: bronzeCount, 'Ignition Zone': ignitionCount });
        historicalScoreData.push({ month, avgScore: Math.round(avgScore * 100) / 100 });
        historicalBadgeData.push({ month, totalBadges });
      }

      setBandTrendData(historicalBandData);
      setScoreTrendData(historicalScoreData);
      setBadgeTrendData(historicalBadgeData);
    } catch (error) {
      console.error('Failed to load historical data:', error);
    }
  };

  const loadAnalytics = async () => {
    setLoading(true);
    try {
      const response = await leaderboardApi.getLeaderboard(selectedMonth, 'all', 'all', '', 0, 1000);
      const managers = response.managers;
      const totalManagers = managers.length;
      const avgScore = managers.reduce((sum, m) => sum + m.finalScore, 0) / totalManagers;

      const goldCount = managers.filter((m) => m.classificationBand === 'Gold').length;
      const silverCount = managers.filter((m) => m.classificationBand === 'Silver').length;
      const bronzeCount = managers.filter((m) => m.classificationBand === 'Bronze').length;
      const ignitionCount = managers.filter((m) => m.classificationBand === 'Ignition Zone').length;

      setStats({ totalManagers, avgScore: Math.round(avgScore * 100) / 100, goldCount, silverCount, bronzeCount, ignitionCount });

      const utilizationRanges = [
        { range: '0-20%', count: 0, color: '#FF6B6B', glow: '#FF6B6B' },
        { range: '21-40%', count: 0, color: '#FFA726', glow: '#FFA726' },
        { range: '41-60%', count: 0, color: '#FFEE58', glow: '#FFEE58' },
        { range: '61-80%', count: 0, color: '#66BB6A', glow: '#66BB6A' },
        { range: '81-100%', count: 0, color: '#00BFA5', glow: '#00BFA5' },
      ];

      managers.forEach((m) => {
        const util = m.utilization;
        if (util <= 20) utilizationRanges[0].count++;
        else if (util <= 40) utilizationRanges[1].count++;
        else if (util <= 60) utilizationRanges[2].count++;
        else if (util <= 80) utilizationRanges[3].count++;
        else utilizationRanges[4].count++;
      });

      setUtilizationData(utilizationRanges.filter((r) => r.count > 0));
    } catch (error) {
      console.error('Failed to load analytics:', error);
    } finally {
      setLoading(false);
    }
  };

  // Band colors map
  const BAND_COLORS = {
    Gold: { main: '#FFD700', glow: '#FFC107' },
    Silver: { main: '#78909C', glow: '#90A4AE' },
    Bronze: { main: '#FF8A65', glow: '#FFAB91' },
    'Ignition Zone': { main: '#EF5350', glow: '#E57373' },
  };

  // Donut data
  const donutData = [
    { name: 'Gold', value: stats.goldCount, color: '#FFD700' },
    { name: 'Silver', value: stats.silverCount, color: '#78909C' },
    { name: 'Bronze', value: stats.bronzeCount, color: '#FF8A65' },
    { name: 'Ignition', value: stats.ignitionCount, color: '#EF5350' },
  ].filter((d) => d.value > 0);

  return (
    <Box sx={{ position: 'relative', minHeight: '100vh' }}>
      {/* ═══ HEADER ═══ */}
      <Box sx={{ mb: 4, animation: `${slideUp} 0.5s ease-out` }}>
        <Box sx={{ display: 'flex', alignItems: 'center', gap: 1.5, mb: 1 }}>
          <InsightsIcon sx={{ fontSize: 36, color: '#00BFA5', animation: `${glow} 3s ease-in-out infinite` }} />
          <Typography
            variant="h4"
            sx={{
              fontWeight: 900,
              background: 'linear-gradient(135deg, #00695C 0%, #00BFA5 50%, #00E5FF 100%)',
              backgroundSize: '200% auto',
              WebkitBackgroundClip: 'text',
              WebkitTextFillColor: 'transparent',
              animation: `${shimmer} 4s linear infinite`,
              letterSpacing: '1px',
            }}
          >
            Analytics Command
          </Typography>
        </Box>
        <Typography sx={{ color: '#00897B', fontWeight: 500, fontSize: '0.95rem' }}>
          Deep-dive into performance metrics, trends & battle-season insights
        </Typography>
      </Box>

      {/* ═══ MONTH SELECTOR ═══ */}
      <Box
        sx={{
          mb: 4,
          display: 'inline-flex',
          alignItems: 'center',
          gap: 2,
          p: 2,
          px: 3,
          borderRadius: 3,
          background: 'linear-gradient(135deg, #0f1629 0%, #1a1f3a 100%)',

      >
        <CalendarIcon sx={{ color: '#00BFA5', fontSize: 22 }} />
        <FormControl size="small" sx={{ minWidth: 200 }}>
          <InputLabel
            sx={{
              color: 'rgba(255,255,255,0.5)',
              '&.Mui-focused': { color: '#00BFA5' },
            }}
          >
            Select Month
          </InputLabel>
          <Select
            value={selectedMonth}
            onChange={(e) => setSelectedMonth(e.target.value)}
            label="Select Month"
            sx={{
              color: '#fff',
              fontWeight: 700,
              '.MuiOutlinedInput-notchedOutline': {

                          <Box

                          sx={{

                            color: 'rgba(255,255,255,0.35)',

                          sx={{
                            p: 2,
                            borderRadius: 2.5,
                            background:
                              'linear-gradient(135deg, rgba(124,77,255,0.1) 0%, rgba(224,64,251,0.05) 100%)',
                            border: '1px solid rgba(124,77,255,0.15)',
                          }}
                        >
                          <Box
                            sx={{

                              <Box
                                key={i}
                                sx={{
                                  width: 24,
                                  height: 6,
                                  borderRadius: 3,
                                  background:
                                    month === selectedMonth
                                      ? '#B388FF'
                                      : 'rgba(179,136,255,0.2)',
                                  boxShadow:
                                    month === selectedMonth
                                      ? '0 0 8px rgba(179,136,255,0.5)'

                                  transition: 'all 0.3s ease',

                      </Box>

                  </Grid>
                </ChartCard>

                  sx={{

                    borderRadius: 4,

                    sx={{
                      display: 'flex',
                      alignItems: 'center',
                      justifyContent: 'space-between',

                    <Box sx={{ display: 'flex', alignItems: 'center', gap: 1.5 }}>
                      <Box
                        sx={{
                          width: 36,

                          borderRadius: 2,

                          alignItems: 'center',
                          justifyContent: 'center',
                          background: 'rgba(0,191,165,0.1)',
                          border: '1px solid rgba(0,191,165,0.2)',
                          fontSize: '1.1rem',
                        }}
                      >
                        🎯
                      </Box>
                      <Typography
                        sx={{
                          fontWeight: 800,
                          color: '#fff',
                          fontSize: '0.95rem',
                        }}
                      >
                        Performance Heatmap
                      </Typography>
                    </Box>
                    <Typography
                      sx={{
                        fontSize: '0.65rem',

                    {[
                      {

                      {

                      {

                        value: stats.silverCount,

                      {
                        label: 'Gold',

                      },
                    ]
                      .filter((b) => b.value > 0)
                      .map((band, i) => {

                          <Box

                              cursor: 'pointer',

                              overflow: 'hidden',

                                inset: 0,

                                  'linear-gradient(180deg, rgba(255,255,255,0.15) 0%, transparent 60%)',

                                fontWeight: 800,
                                color: 'rgba(0,0,0,0.7)',
                                textTransform: 'uppercase',

                                zIndex: 1,

                              {pct >= 15

                  {/* Heatmap legend */}

                    sx={{

                      { label: 'Bronze', color: '#FF8A65', value: stats.bronzeCount },

                      { label: 'Gold', color: '#FFD700', value: stats.goldCount },
                    ].map((item, i) => (
                      <Box
                        key={i}
                        sx={{ display: 'flex', alignItems: 'center', gap: 0.8 }}
                      >
                        <Box
                          sx={{
                            width: 10,
                            height: 10,
                            borderRadius: '50%',
                            background: item.color,
                            boxShadow: `0 0 6px ${item.color}60`,

                        </Typography>
                        <Typography
                          sx={{
                            fontSize: '0.72rem',

                        </Typography>
                      </Box>
                    ))}
                  </Box>

          )}
        </>
      )}
